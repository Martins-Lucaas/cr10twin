/* ============================================================
 * TEMPLATE
 * ============================================================ */

void init_template(void)
{
    g_template[0] = 0.0f;

    for (int i = 1;
         i < TEMPLATE_SIZE;
         i++)
    {
        g_template[i] =
            G_MAX *
            expf(
                -(float)(i - 1) /
                4.0f
            );
    }
}


/* ============================================================
 * USB BUFFER
 * ============================================================ */

uint16_t usb_buffer_free_space(void)
{
    uint16_t head = usb_head;
    uint16_t tail = usb_tail;

    if (head >= tail)
    {
        return
            (USB_TX_BUFFER_SIZE - 1U)
            - (head - tail);
    }

    return
        tail - head - 1U;
}


/* ============================================================ */

bool usb_buffer_write_atomic(
    const char *data,
    uint16_t len
)
{
    if (len == 0U)
    {
        return true;
    }

    /*
     * So grava se a mensagem inteira couber.
     */
    if (usb_buffer_free_space() < len)
    {
        return false;
    }

    for (uint16_t i = 0;
         i < len;
         i++)
    {
        usb_tx_buffer[usb_head] =
            (uint8_t)data[i];

        usb_head =
            (usb_head + 1U)
            % USB_TX_BUFFER_SIZE;
    }

    return true;
}


/* ============================================================ */

void usb_buffer_process(void)
{
    static uint8_t
    packet[USB_PACKET_SIZE];

    uint16_t len = 0;

    uint16_t temp_tail =
        usb_tail;


    while (
        (temp_tail != usb_head) &&
        (len < USB_PACKET_SIZE)
    )
    {
        packet[len++] =
            usb_tx_buffer[temp_tail];

        temp_tail =
            (temp_tail + 1U)
            % USB_TX_BUFFER_SIZE;
    }


    if (
        (len > 0U) &&
        (hUsbDeviceFS.dev_state ==
         USBD_STATE_CONFIGURED)
    )
    {
        if (
            CDC_Transmit_FS(
                packet,
                len
            ) == USBD_OK
        )
        {
            usb_tail =
                temp_tail;
        }
    }
}


/* ============================================================
 * SELECAO DAS LINHAS
 *
 * LINHAS COMPARTILHADAS PELOS DOIS SENSORES
 *
 * L0 -> PD0
 * L1 -> PD1
 * L2 -> PF3
 * L3 -> PF5
 * L4 -> PF10
 *
 * Uma linha fica LOW
 * e as outras ficam HIGH.
 * ============================================================ */

const uint8_t row_masks[ROWS] =
{
    0b11110,
    0b11101,
    0b11011,
    0b10111,
    0b01111
};


/* ============================================================ */

void select_row(uint8_t row)
{
    if (row >= ROWS)
    {
        return;
    }

    uint8_t mask =
        row_masks[row];


    /*
     * L0 = PD0
     */
    HAL_GPIO_WritePin(
        GPIOD,
        GPIO_PIN_0,
        (mask & (1U << 0))
            ? GPIO_PIN_SET
            : GPIO_PIN_RESET
    );


    /*
     * L1 = PD1
     */
    HAL_GPIO_WritePin(
        GPIOD,
        GPIO_PIN_1,
        (mask & (1U << 1))
            ? GPIO_PIN_SET
            : GPIO_PIN_RESET
    );


    /*
     * L2 = PF3
     */
    HAL_GPIO_WritePin(
        GPIOF,
        GPIO_PIN_3,
        (mask & (1U << 2))
            ? GPIO_PIN_SET
            : GPIO_PIN_RESET
    );


    /*
     * L3 = PF5
     */
    HAL_GPIO_WritePin(
        GPIOF,
        GPIO_PIN_5,
        (mask & (1U << 3))
            ? GPIO_PIN_SET
            : GPIO_PIN_RESET
    );


    /*
     * L4 = PF10
     */
    HAL_GPIO_WritePin(
        GPIOF,
        GPIO_PIN_10,
        (mask & (1U << 4))
            ? GPIO_PIN_SET
            : GPIO_PIN_RESET
    );
}


/* ============================================================
 * IZHIKEVICH
 * ============================================================ */

bool izhikevich_step(
    float *v,
    float *u,
    float I,
    float a,
    float b,
    float c,
    float d
)
{
    *v +=
        DT *
        (
            0.04f *
            (*v) *
            (*v)

            + 5.0f *
            (*v)

            + 140.0f

            - (*u)

            + I
        );


    *u +=
        DT *
        (
            a *
            (
                b * (*v)
                - (*u)
            )
        );


    if (*v >= VTH)
    {
        *v = c;

        *u += d;

        return true;
    }

    return false;
}


/* ============================================================
 * PROCESSAMENTO DOS TAXELS
 *
 * RA e SA sao calculados internamente.
 * Eles alimentam o neuronio CN_MM.
 * ============================================================ */

void update_taxels(
    uint8_t sensor,
    Taxel *t,
    uint16_t *adc,
    uint8_t row_idx
)
{
    HAL_GPIO_WritePin(
        DEBUG_IZH_PORT,
        DEBUG_IZH_PIN,
        GPIO_PIN_SET
    );


    for (int i = 0;
         i < COLS;
         i++)
    {
        int global_idx =
            row_idx * COLS + i;


        /* ====================================================
         * ADC -> TENSAO
         * ==================================================== */

        float V =
            adc[i] *
            (V_MAX / 4095.0f);


        float Vn =
            (V_MAX - V) /
            (V_MAX - V_MIN);


        float I_raw =
            Vn;


        /* ====================================================
         * DERIVADA PARA RA
         * ==================================================== */

        uint8_t idx =
            I_index[sensor][global_idx];


        float I_old =
            I_buffer
            [sensor]
            [global_idx]
            [idx];


        I_buffer
        [sensor]
        [global_idx]
        [idx] =
            I_raw;


        I_index
        [sensor]
        [global_idx] =
            (idx + 1U)
            % DIFF_BUFFER;


        float dI =
            fabsf(
                I_raw -
                I_old
            )
            /
            (DIFF_BUFFER * DT);


        if (fabsf(dI) < 0.01f)
        {
            dI = 0.0f;
        }


        float I_RA =
            G_RA *
            fabsf(dI);


        float I_SA =
            G_SA *
            I_raw;


        /* ====================================================
         * RA
         * ==================================================== */

        bool spike_ra =
            izhikevich_step(
                &t[i].v_RA,
                &t[i].u_RA,
                I_RA,
                A_RA,
                B_RA,
                C_RA,
                D_RA
            );


        if (spike_ra)
        {
            template_idx_RA
            [sensor]
            [global_idx] = 1;


            template_idx_INH_RA
            [sensor]
            [global_idx] = 1;
        }


        /* ====================================================
         * SA
         * ==================================================== */

        bool spike_sa =
            izhikevich_step(
                &t[i].v_SA,
                &t[i].u_SA,
                I_SA,
                A_SA,
                B_SA,
                C_SA,
                D_SA
            );


        if (spike_sa)
        {
            template_idx_SA
            [sensor]
            [global_idx] = 1;


            template_idx_INH_SA
            [sensor]
            [global_idx] = 1;
        }


        /* ====================================================
         * SINAPSE EXCITATORIA RA
         * ==================================================== */

        float I_ra = 0.0f;


        uint8_t idx_ra =
            template_idx_RA
            [sensor]
            [global_idx];


        if (idx_ra > 0U)
        {
            I_ra =
                gain_RA
                [sensor]
                [global_idx]
                *
                g_template[idx_ra];


            idx_ra++;


            if (
                idx_ra >=
                TEMPLATE_SIZE
            )
            {
                idx_ra = 0;
            }


            template_idx_RA
            [sensor]
            [global_idx] =
                idx_ra;
        }


        /* ====================================================
         * SINAPSE EXCITATORIA SA
         * ==================================================== */

        float I_sa = 0.0f;


        uint8_t idx_sa =
            template_idx_SA
            [sensor]
            [global_idx];


        if (idx_sa > 0U)
        {
            I_sa =
                gain_SA
                [sensor]
                [global_idx]
                *
                g_template[idx_sa];


            idx_sa++;


            if (
                idx_sa >=
                TEMPLATE_SIZE
            )
            {
                idx_sa = 0;
            }


            template_idx_SA
            [sensor]
            [global_idx] =
                idx_sa;
        }


        /* ====================================================
         * MULTIMODAL RECEBE RA + SA
         * ==================================================== */

        I_total_MM[sensor] +=
            I_ra;

        I_total_MM[sensor] +=
            I_sa;


        /* Salva ADC */
        last_adc
        [sensor]
        [global_idx] =
            adc[i];
    }


    HAL_GPIO_WritePin(
        DEBUG_IZH_PORT,
        DEBUG_IZH_PIN,
        GPIO_PIN_RESET
    );
}


/* ============================================================
 * NEURONIO CUNEIFORME MULTIMODAL
 * ============================================================ */

void process_cuneate(
    uint8_t sensor,
    uint32_t tstamp
)
{
    float I_inh =
        0.0f;


    /* ========================================================
     * INIBICAO RA
     * ======================================================== */

    for (int k = 0;
         k < NUM_TAXELS;
         k++)
    {
        uint8_t idx_inh_ra =
            template_idx_INH_RA
            [sensor]
            [k];


        if (idx_inh_ra > 0U)
        {
            I_inh -=
                W_INB *
                g_template[idx_inh_ra];


            idx_inh_ra++;


            if (
                idx_inh_ra >=
                TEMPLATE_SIZE
            )
            {
                idx_inh_ra = 0;
            }


            template_idx_INH_RA
            [sensor]
            [k] =
                idx_inh_ra;
        }
    }


    /* ========================================================
     * INIBICAO SA
     * ======================================================== */

    for (int k = 0;
         k < NUM_TAXELS;
         k++)
    {
        uint8_t idx_inh_sa =
            template_idx_INH_SA
            [sensor]
            [k];


        if (idx_inh_sa > 0U)
        {
            I_inh -=
                W_INB *
                g_template[idx_inh_sa];


            idx_inh_sa++;


            if (
                idx_inh_sa >=
                TEMPLATE_SIZE
            )
            {
                idx_inh_sa = 0;
            }


            template_idx_INH_SA
            [sensor]
            [k] =
                idx_inh_sa;
        }
    }


    /* ========================================================
     * NORMALIZACAO
     * ======================================================== */

    float I_MM_norm =
        I_total_MM[sensor] /
        (NUM_TAXELS * 2.0f);


    float I_final_MM =
        (I_MM_norm + I_inh)
        *
        G_CN;


    bool spike_mm =
        izhikevich_step(
            &CN[sensor].v_CN,
            &CN[sensor].u_CN,
            I_final_MM,
            A_CN,
            B_CN,
            C_CN,
            D_CN
        );


    /*
     * Nao faz snprintf na interrupcao.
     * Apenas marca o spike.
     */
    if (spike_mm)
    {
        cn_mm_spike_timestamp[sensor] =
            tstamp;

        cn_mm_spike_pending[sensor] =
            1;
    }


    I_total_MM[sensor] =
        0.0f;
}

            adc_frame[buffer_idx][1][16],

            (unsigned)
            adc_frame[buffer_idx][1][17],

            (unsigned)
            adc_frame[buffer_idx][1][18],

            (unsigned)
            adc_frame[buffer_idx][1][19],

            (unsigned)
            adc_frame[buffer_idx][1][20],

            (unsigned)
            adc_frame[buffer_idx][1][21],

            (unsigned)
            adc_frame[buffer_idx][1][22],

            (unsigned)
            adc_frame[buffer_idx][1][23],

            (unsigned)
            adc_frame[buffer_idx][1][24],


            (unsigned long)
            tstamp
        );


    if (
        (n <= 0) ||
        ((size_t)n >= sizeof(msg))
    )
    {
        return false;
    }


    return
        usb_buffer_write_atomic(
            msg,
            (uint16_t)n
        );
}


/* ============================================================
 * ENVIO DOS SPIKES CN_MM
 * ============================================================ */

void send_pending_cn_mm(void)
{
    for (
        uint8_t sensor = 0;
        sensor < NUM_SENSORS;
        sensor++
    )
    {
        if (
            cn_mm_spike_pending[sensor]
            == 0U
        )
        {
            continue;
        }


        uint32_t timestamp;


        __disable_irq();

        timestamp =
            cn_mm_spike_timestamp[sensor];

        __enable_irq();


        char msg[64];

        int n;


        if (sensor == 0U)
        {
            n =
                snprintf(
                    msg,
                    sizeof(msg),

                    "MM1=1,t=%lu\r\n",

                    (unsigned long)
                    timestamp
                );
        }
        else
        {
            n =
                snprintf(
                    msg,
                    sizeof(msg),

                    "MM2=1,t=%lu\r\n",

                    (unsigned long)
                    timestamp
                );
        }


        if (
            (n > 0) &&
            ((size_t)n < sizeof(msg))
        )
        {
            if (
                usb_buffer_write_atomic(
                    msg,
                    (uint16_t)n
                )
            )
            {
                __disable_irq();

                cn_mm_spike_pending[sensor] =
                    0U;

                __enable_irq();
            }
        }
    }
}


/* ============================================================
 * ADC CALLBACK
 *
 * DMA:
 *
 * adc_buffer[0..4] = SENSOR 1
 * adc_buffer[5..9] = SENSOR 2
 * ============================================================ */

void HAL_ADC_ConvCpltCallback(
    ADC_HandleTypeDef *hadc
)
{
    if (hadc != &hadc1)
    {
        return;
    }


    HAL_GPIO_WritePin(
        DEBUG_ADC_PORT,
        DEBUG_ADC_PIN,
        GPIO_PIN_SET
    );


    /* ========================================================
     * SENSOR 1
     *
     * adc_buffer[0..4]
     * ======================================================== */

    update_taxels(
        0,

        &taxels
        [0]
        [current_row * COLS],

        &adc_buffer[0],

        current_row
    );


    /* ========================================================
     * SENSOR 2
     *
     * adc_buffer[5..9]
     * ======================================================== */

    update_taxels(
        1,

        &taxels
        [1]
        [current_row * COLS],

        &adc_buffer[COLS],

        current_row
    );


    /* ========================================================
     * TERMINOU AS 5 LINHAS
     * ======================================================== */

    if (
        current_row ==
        (ROWS - 1)
    )
    {
        uint32_t tstamp =
            __HAL_TIM_GET_COUNTER(
                &htim2
            );


        /*
         * Um CN_MM para cada sensor
         */
        process_cuneate(
            0,
            tstamp
        );

        process_cuneate(
            1,
            tstamp
        );


        /*
         * Salva frame ADC completo
         */
        uint8_t wb =
            adc_write_buffer;


        for (
            int sensor = 0;
            sensor < NUM_SENSORS;
            sensor++
        )
        {
            for (
                int i = 0;
                i < NUM_TAXELS;
                i++
            )
            {
                adc_frame
                [wb]
                [sensor]
                [i] =

                    last_adc
                    [sensor]
                    [i];
            }
        }


        adc_frame_timestamp[wb] =
            tstamp;


        adc_ready_buffer =
            wb;


        adc_frame_ready =
            1;


        adc_write_buffer =
            wb ^ 1U;
    }


    /*
     * Seleciona proxima linha
     */
    current_row =
        (current_row + 1U)
        % ROWS;


    select_row(
        current_row
    );


    HAL_GPIO_WritePin(
        DEBUG_ADC_PORT,
        DEBUG_ADC_PIN,
        GPIO_PIN_RESET
    );
}


/* ============================================================
 * MAIN
 * ============================================================ */

int main(void)
{
    HAL_Init();


    SystemClock_Config();


    MX_GPIO_Init();

    MX_DMA_Init();

    MX_ADC1_Init();

    MX_TIM6_Init();

    MX_TIM2_Init();

    MX_USB_DEVICE_Init();


    init_template();


    /* ========================================================
     * INICIALIZA CN_MM
     * ======================================================== */

    for (
        int sensor = 0;
        sensor < NUM_SENSORS;
        sensor++
    )
    {
        CN[sensor].v_CN =
            -30.0f;


        CN[sensor].u_CN =
            B_CN *
            CN[sensor].v_CN;
    }


    /* ========================================================
     * INICIALIZA OS 50 TAXELS
     * ======================================================== */

    for (
        int sensor = 0;
        sensor < NUM_SENSORS;
        sensor++
    )
    {
        for (
            int i = 0;
            i < NUM_TAXELS;
            i++
        )
        {
            taxels[sensor][i].v_RA =
                -30.0f;


            taxels[sensor][i].u_RA =
                B_RA *
                taxels[sensor][i].v_RA;


            taxels[sensor][i].v_SA =
                -30.0f;


            taxels[sensor][i].u_SA =
                B_SA *
                taxels[sensor][i].v_SA;


            taxels[sensor][i].I =
                0.0f;


            gain_RA[sensor][i] =
                gain_RA_init[i];


            gain_SA[sensor][i] =
                gain_SA_init[i];
        }
    }


    /* ========================================================
     * PRIMEIRA LINHA
     * ======================================================== */

    current_row = 0;

    select_row(
        current_row
    );


    /* ========================================================
     * TIMESTAMP
     * ======================================================== */

    HAL_TIM_Base_Start(
        &htim2
    );


    /* ========================================================
     * ADC + DMA
     *
     * 10 conversoes:
     *
     * S1 = 5
     * S2 = 5
     * ======================================================== */

    if (
        HAL_ADC_Start_DMA(
            &hadc1,

            (uint32_t *)
            adc_buffer,

            ADC_CHANNELS
        )
        != HAL_OK
    )
    {
        Error_Handler();
    }


    /* ========================================================
     * INICIA TRIGGER DO ADC
     * ======================================================== */

    if (
        HAL_TIM_Base_Start(
            &htim6
        )
        != HAL_OK
    )
    {
        Error_Handler();
    }


    const char boot_msg[] =

        "BOOT OK - "
        "S1+S2 ADC / "
        "MM EVENT ONLY\r\n";


    usb_buffer_write_atomic(
        boot_msg,

        (uint16_t)
        (sizeof(boot_msg) - 1U)
    );


    /* ========================================================
     * LOOP PRINCIPAL
     * ======================================================== */

    while (1)
    {
        /*
         * Drena USB
         */
        usb_buffer_process();


        /*
         * Envia MM1/MM2
         */
        send_pending_cn_mm();


        /*
         * Envia ultimo frame ADC
         */
        if (adc_frame_ready)
        {
            uint8_t buffer_idx;

            uint32_t timestamp;


            __disable_irq();


            buffer_idx =
                adc_ready_buffer;


            timestamp =
                adc_frame_timestamp
                [buffer_idx];


            adc_frame_ready =
                0;


            __enable_irq();


            send_adc_frame_both(
                buffer_idx,
                timestamp
            );
        }


        usb_buffer_process();
    }
}

/* ============================================================
 * ADC1
 *
 * TOTAL = 10 CONVERSOES
 * ============================================================ */

void MX_ADC1_Init(void)
{
    __HAL_RCC_ADC1_CLK_ENABLE();


    ADC_ChannelConfTypeDef c = {0};


    hadc1.Instance =
        ADC1;


    hadc1.Init.Resolution =
        ADC_RESOLUTION_12B;


    hadc1.Init.ScanConvMode =
        ENABLE;


    hadc1.Init.ContinuousConvMode =
        DISABLE;


    hadc1.Init.NbrOfConversion =
        ADC_CHANNELS;


    hadc1.Init.ExternalTrigConv =
        ADC_EXTERNALTRIGCONV_T6_TRGO;


    hadc1.Init.ExternalTrigConvEdge =
        ADC_EXTERNALTRIGCONVEDGE_RISING;


    hadc1.Init.DMAContinuousRequests =
        ENABLE;


    if (
        HAL_ADC_Init(
            &hadc1
        )
        != HAL_OK
    )
    {
        Error_Handler();
    }


    /*
     * ========================================================
     * ORDEM DO DMA
     * ========================================================
     *
     * SENSOR 1
     *
     * [0] PA0 = ADC_CHANNEL_0
     * [1] PA3 = ADC_CHANNEL_3
     * [2] PA4 = ADC_CHANNEL_4
     * [3] PA6 = ADC_CHANNEL_6
     * [4] PB1 = ADC_CHANNEL_9
     *
     *
     * SENSOR 2
     *
     * [5] PA5 = ADC_CHANNEL_5
     * [6] PA7 = ADC_CHANNEL_7
     * [7] PC0 = ADC_CHANNEL_10
     * [8] PC3 = ADC_CHANNEL_13
     * [9] PC2 = ADC_CHANNEL_12
     *
     * ========================================================
     */


    uint32_t ch[ADC_CHANNELS] =
    {
        /* ================================================
         * SENSOR 1
         * ================================================ */

        ADC_CHANNEL_0,     /* PA0 */

        ADC_CHANNEL_3,     /* PA3 */

        ADC_CHANNEL_4,     /* PA4 */

        ADC_CHANNEL_6,     /* PA6 */

        ADC_CHANNEL_9,     /* PB1 */


        /* ================================================
         * SENSOR 2
         * ================================================ */

        ADC_CHANNEL_5,     /* PA5 */

        ADC_CHANNEL_8,     /* PA7 */

        ADC_CHANNEL_10,    /* PC0 */

        ADC_CHANNEL_13,    /* PC3 */

        ADC_CHANNEL_12     /* PC2 */
    };


    for (
        int i = 0;
        i < ADC_CHANNELS;
        i++
    )
    {
        c.Channel =
            ch[i];


        c.Rank =
            i + 1;


        c.SamplingTime =
            ADC_SAMPLETIME_15CYCLES;


        if (
            HAL_ADC_ConfigChannel(
                &hadc1,
                &c
            )
            != HAL_OK
        )
        {
            Error_Handler();
        }
    }
}
